                \midrule
                Power-matched stealth & receiver (TEXBAT) \\
                Carrier-phase rotation & receiver (TEXBAT) \\
                Code-drift / time push & receiver (TEXBAT) \\
                Per-channel spoof & UAV (Aissou) \\
                Teleport (skok lat/lon) & aviation (ADS-B) \\
                Slow drift & aviation (ADS-B) \\
                Frozen position & aviation (ADS-B) \\
                Implausible velocity & aviation (ADS-B) \\
                \bottomrule
            \end{tabular}
        \end{block}

        \vspace{0.4em}
        \begin{alertblock}{Luka w obronie}
            \scriptsize CRPA, INS, OSNMA — każde wymaga \textbf{lat} do wdrożenia. Potrzebujemy obrony software'owej \textbf{na istniejącym sprzęcie, dziś}.
        \end{alertblock}

% SLIDE 4 — Feature Engineering (Criterion 3)
% =========================================================
\begin{frame}{Feature Engineering — 4 rodziny features}
    \criterion{3: Feature Engineering / Understanding}
    \vspace{0.15cm}
    \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
        \footnotesize
        \begin{enumerate}\setlength\itemsep{0.35em}
            \item \textbf{Multi-correlator SQM} — 41 I + 41 Q taps autocorrelation function. Łapie carrier phase distortion. \textit{Klasyka literatury (Wesson, Phelts).}
            \begin{itemize}
                \item \textcolor{accent}{\textbf{`secondary\_peak\_ratio`}} — \textbf{nasz custom feature}: max envelope poza ±5 tapów od peak'u
            \end{itemize}
            \item \textbf{Wideband AGC power} (2/4/8 MHz) — spoof transmitter zwiększa total RF power, AGC reaguje sub-dB
            \item \textbf{NavSol residuals} — clock drift + position drift (walk phase)
            \item \textbf{Trajectory kinematics} — haversine step distance, velocity-position residuals, heading changes (L3)
        \end{enumerate}

        \vspace{0.3em}
        \begin{block}{Custom: Z-score normalization}
            \scriptsize Per-scenariusz baseline t<100s — adresuje hardware drift między recording sessions. \textbf{Bez tego: F1 spada z 0.984 → 0.17.}
        \end{block}
    \column{0.45\textwidth}
        \centering
        \includegraphics[width=\linewidth]{../docs/highroi/03_shap_global_l1.png}\\[0.2em]
        {\scriptsize\color{black!60} L1 SHAP global: top features na ds7 OOD — \texttt{power\_2MHz\_zscore}, \texttt{sqm\_asymmetry}, \texttt{secondary\_peak\_ratio} dominują}
    \end{columns}
\end{frame}

\begin{frame}{Interpretation \& Limitations — uczciwa analiza}
    \criterion{8: Interpretation of Results}
    \vspace{0.1cm}
    \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
        \begin{alertblock}{3 zidentyfikowane słabości}
            \footnotesize
            \begin{enumerate}\setlength\itemsep{0.3em}
                \item \textbf{L3v2 calibration problem (zaadresowany)} \\
                \scriptsize Pierwotny F1=0.74 był zawyżony — FPR=51\% na clean. Po retreningu na 2× datasecie (Faza 11): uczciwa kalibracja FPR=10\%, recall=25\% przy operacyjnym thresholdzie.
                \item \textbf{Drift attack <1 km/min nieuwykrywalny} \\
                \scriptsize Z samej pozycji+prędkości — poniżej GPS noise floor. Roadmap: INS fusion, multi-receiver TDOA.
                \item \textbf{Receiver bias TEXBAT} \\
                \scriptsize Wszystkie 4 scenariusze z UT GRID rig. Generalizacja na inny hardware untested.
            \end{enumerate}
        \end{alertblock}

        \begin{block}{Trade-off przy operacyjnym progu}
            \scriptsize \textbf{Precision=1.000 (zero FP) > high recall}. FP = lądowanie awaryjne; FN = istniejący monitoring zostaje aktywny.
        \end{block}
